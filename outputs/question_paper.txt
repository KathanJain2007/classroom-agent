Question 7. [6 Marks]
Language: C
The program reads an integer n, allocates an array of n ints, fills it, then attempts to double its size using realloc to store additional values. Identify the flaw in the realloc usage and provide a corrected version.

Code Snippet:
    #include <stdio.h>
    #include <stdlib.h>
    
    int main(){
        int n; scanf("%d",&n);
        int *arr = malloc(n * sizeof(int));
        if(!arr){ perror("malloc"); return 1; }
        for(int i=0;i<n;i++) arr[i]=i+1;
        /* attempt to expand */
        arr = realloc(arr, 2*n * sizeof(int)); // BUG: no check for failure
        if(arr==NULL){ perror("realloc"); return 1; }
        for(int i=n;i<2*n;i++) arr[i]=i+1;
        for(int i=0;i<2*n;i++) printf("%d ",arr[i]);
        free(arr);
        return 0;
    }

Question 8. [8 Marks]
Language: C
The following C program implements a fixed‑size queue using an array of size 5. It provides enqueue and dequeue functions, but it contains logical errors that cause incorrect overflow/underflow detection and wrong front/rear updates. Identify and correct the errors so that the queue works correctly for the sequence of operations shown in main.

Code Snippet:
    #include <stdio.h>
    #define MAX 5
    int queue[MAX];
    int front = -1;
    int rear = -1;
    
    int isFull() {
        return rear == MAX; // BUG: should be MAX-1
    }
    
    int isEmpty() {
        return front == -1 || front > rear;
    }
    
    void enqueue(int x) {
        if (isFull()) {
            printf("Overflow\n");
            return;
        }
        if (front == -1) front = 0;
        rear++;
        queue[rear] = x;
    }
    
    int dequeue() {
        if (isEmpty()) {
            printf("Underflow\n");
            return -1;
        }
        int val = queue[front];
        front++;
        if (front > rear) {
            front = rear = -1; // reset queue
        }
        return val;
    }
    
    int main() {
        enqueue(10);
        enqueue(20);
        enqueue(30);
        enqueue(40);
        enqueue(50);
        enqueue(60); // should trigger overflow
        printf("%d ", dequeue());
        printf("%d ", dequeue());
        enqueue(70);
        enqueue(80); // should trigger overflow again
        while (!isEmpty()) {
            printf("%d ", dequeue());
        }
        dequeue(); // should trigger underflow
        return 0;
    }

Question 9. [8 Marks]
Language: C
The following C program implements an array‑based deque of maximum size 10. It provides functions insertRear, insertFront, deleteRear, deleteFront and display. However, the insertFront function does not correctly handle the case when the front index is at position 0, causing an out‑of‑bounds write. Identify and correct the bug so that the deque works as a circular array.

Code Snippet:
    #include <stdio.h>
    #define MAX 10
    
    int deque[MAX];
    int front = -1;
    int rear = -1;
    
    int isFull() {
        return (front == 0 && rear == MAX-1) || (front == rear+1);
    }
    
    int isEmpty() {
        return front == -1;
    }
    
    void insertRear(int x) {
        if (isFull()) {
            printf("Overflow\n");
            return;
        }
        if (isEmpty()) {
            front = rear = 0;
        } else if (rear == MAX-1) {
            rear = 0;
        } else {
            rear++;
        }
        deque[rear] = x;
    }
    
    void insertFront(int x) {
        if (isFull()) {
            printf("Overflow\n");
            return;
        }
        if (isEmpty()) {
            front = rear = 0;
        } else if (front == 0) {
            front = MAX;   /* BUG: should wrap to MAX-1 */
        } else {
            front--;
        }
        deque[front] = x;
    }
    
    int deleteFront() {
        if (isEmpty()) {
            printf("Underflow\n");
            return -1;
        }
        int data = deque[front];
        if (front == rear) {
            front = rear = -1;
        } else if (front == MAX-1) {
            front = 0;
        } else {
            front++;
        }
        return data;
    }
    
    int deleteRear() {
        if (isEmpty()) {
            printf("Underflow\n");
            return -1;
        }
        int data = deque[rear];
        if (front == rear) {
            front = rear = -1;
        } else if (rear == 0) {
            rear = MAX-1;
        } else {
            rear--;
        }
        return data;
    }
    
    void display() {
        if (isEmpty()) {
            printf("Deque is empty\n");
            return;
        }
        int i = front;
        printf("Deque contents: ");
        while (1) {
            printf("%d ", deque[i]);
            if (i == rear) break;
            i = (i + 1) % MAX;
        }
        printf("\n");
    }
    
    int main() {
        insertRear(10);
        insertRear(20);
        insertFront(5);
        insertFront(2);
        display();
        deleteRear();
        deleteFront();
        display();
        return 0;
    }
